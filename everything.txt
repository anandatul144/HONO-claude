# HONO IDS SOC Docker Compose Configuration
# This configuration implements a minimal HONO 2.6 stack for bridging SOVD Client and Server
# Architecture: SOVD Server (Device) <-> HONO HTTP Adapter <-> SOVD Client
# Security: Insecure mode for development (no authentication required)

version: '3.8'

services:
  # =============================================================================
  # HONO Core Services - Minimal Stack
  # =============================================================================
  
  # Device Registry Service
  # Manages device registration and metadata
  # Exposes HTTP API for device management and registration UI
  hono-device-registry:
    image: eclipse/hono-service-device-registry:2.6.0
    container_name: hono-device-registry
    ports:
      - "28080:8080"  # HTTP Management API
      - "28443:8443"  # HTTPS Management API (unused in insecure mode)
    environment:
      # Service configuration
      HONO_REGISTRY_SVC_BIND_ADDRESS: "0.0.0.0"
      HONO_REGISTRY_SVC_INSECURE_PORT: "5671"
      HONO_REGISTRY_SVC_INSECURE_PORT_ENABLED: "true"
      
      # HTTP endpoint configuration for device registration
      HONO_REGISTRY_HTTP_BIND_ADDRESS: "0.0.0.0"
      HONO_REGISTRY_HTTP_PORT: "8080"
      HONO_REGISTRY_HTTP_INSECURE_PORT_ENABLED: "true"
      
      # Disable authentication for development
      HONO_REGISTRY_SVC_AUTH_VALIDATION_SHARED_SECRET: "dummy-secret"
      HONO_REGISTRY_CREDENTIALS_SVC_CREDENTIALS_TTL: "1h"
      
      # Enable device auto-provisioning
      HONO_REGISTRY_SVC_RECEIVER_LINK_CREDIT: "100"
      
      # Logging configuration
      HONO_LOG_LEVEL: "INFO"
      
    volumes:
      - hono-device-registry-data:/var/lib/hono/device-registry
    networks:
      - hono-network
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8080/health"]
      interval: 30s
      timeout: 10s
      retries: 3

  # HTTP Protocol Adapter
  # Handles HTTP communication between SOVD devices and clients
  # Enables RESTful API for device command and control
  hono-adapter-http:
    image: eclipse/hono-adapter-http:2.6.0
    container_name: hono-adapter-http
    depends_on:
      - hono-device-registry
      - kafka
    ports:
      - "8080:8080"   # HTTP port for device communication
      - "8443:8443"   # HTTPS port (unused in insecure mode)
    environment:
      # HTTP adapter configuration
      HONO_HTTP_BIND_ADDRESS: "0.0.0.0"
      HONO_HTTP_PORT: "8080"
      HONO_HTTP_INSECURE_PORT_ENABLED: "true"
      
      # Disable authentication for development
      HONO_HTTP_AUTHENTICATION_REQUIRED: "false"
      HONO_HTTP_TENANT_IDLE_TIMEOUT: "0s"
      
      # Device registry connection
      HONO_REGISTRY_HOST: "hono-device-registry"
      HONO_REGISTRY_PORT: "5671"
      HONO_REGISTRY_USERNAME: "consumer@HONO"
      HONO_REGISTRY_PASSWORD: "verysecret"
      
      # Kafka message broker connection
      HONO_KAFKA_BOOTSTRAP_SERVERS: "kafka:29092"
      HONO_KAFKA_PRODUCER_CONFIG: "acks=1,retries=3,batch.size=16384"
      HONO_KAFKA_CONSUMER_CONFIG: "enable.auto.commit=true,auto.commit.interval.ms=1000"
      
      # Message routing configuration
      HONO_MESSAGING_HOST: "kafka"
      HONO_MESSAGING_PORT: "29092"
      HONO_MESSAGING_TYPE: "kafka"
      
      # Command and control via Kafka
      HONO_COMMAND_HOST: "kafka"
      HONO_COMMAND_PORT: "29092"
      HONO_COMMAND_TYPE: "kafka"
      
      # Logging
      HONO_LOG_LEVEL: "INFO"
      
    networks:
      - hono-network
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8080/health"]
      interval: 30s
      timeout: 10s
      retries: 3

  # Apache Kafka Message Broker
  # Message routing between HONO components and SOVD clients
  # Handles event streaming and message brokering for IDS SOC
  kafka:
    image: confluentinc/cp-kafka:7.5.0
    container_name: hono-kafka
    depends_on:
      - zookeeper
    ports:
      - "9092:9092"   # Kafka broker port
      - "9093:9093"   # External listener port
    environment:
      # Kafka broker configuration
      KAFKA_BROKER_ID: 1
      KAFKA_ZOOKEEPER_CONNECT: "zookeeper:2181"
      
      # Listener configuration for both internal and external access
      KAFKA_LISTENERS: "INTERNAL://0.0.0.0:29092,EXTERNAL://0.0.0.0:9092"
      KAFKA_ADVERTISED_LISTENERS: "INTERNAL://kafka:29092,EXTERNAL://localhost:9092"
      KAFKA_LISTENER_SECURITY_PROTOCOL_MAP: "INTERNAL:PLAINTEXT,EXTERNAL:PLAINTEXT"
      KAFKA_INTER_BROKER_LISTENER_NAME: "INTERNAL"
      
      # Topic configuration
      KAFKA_AUTO_CREATE_TOPICS_ENABLE: "true"
      KAFKA_NUM_PARTITIONS: 3
      KAFKA_DEFAULT_REPLICATION_FACTOR: 1
      
      # Performance and retention settings
      KAFKA_OFFSETS_TOPIC_REPLICATION_FACTOR: 1
      KAFKA_LOG_RETENTION_HOURS: 24
      KAFKA_LOG_RETENTION_BYTES: 1073741824  # 1GB
      
      # JVM heap settings
      KAFKA_HEAP_OPTS: "-Xmx512m -Xms512m"
      
    volumes:
      - kafka-data:/var/lib/kafka/data
    networks:
      - hono-network
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "kafka-broker-api-versions", "--bootstrap-server", "localhost:9092"]
      interval: 30s
      timeout: 10s
      retries: 3

  # Apache Zookeeper
  # Required for Kafka cluster coordination
  zookeeper:
    image: confluentinc/cp-zookeeper:7.5.0
    container_name: hono-zookeeper
    ports:
      - "2181:2181"
    environment:
      ZOOKEEPER_CLIENT_PORT: 2181
      ZOOKEEPER_TICK_TIME: 2000
      ZOOKEEPER_INIT_LIMIT: 10
      ZOOKEEPER_SYNC_LIMIT: 5
    volumes:
      - zookeeper-data:/var/lib/zookeeper/data
      - zookeeper-logs:/var/lib/zookeeper/log
    networks:
      - hono-network
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "echo", "ruok", "|", "nc", "localhost", "2181"]
      interval: 30s
      timeout: 10s
      retries: 3

  # =============================================================================
  # Supporting Services
  # =============================================================================
  
  # MongoDB for SOVD Client data storage
  # Stores JSON responses from SOVD devices
  mongodb:
    image: mongo:7.0
    container_name: hono-mongodb
    ports:
      - "27017:27017"
    environment:
      MONGO_INITDB_ROOT_USERNAME: "admin"
      MONGO_INITDB_ROOT_PASSWORD: "password"
      MONGO_INITDB_DATABASE: "sovd_data"
    volumes:
      - mongodb-data:/data/db
      - "./config/mongo-init.js:/docker-entrypoint-initdb.d/mongo-init.js:ro"
    networks:
      - hono-network
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "mongosh", "--eval", "db.adminCommand('ping')"]
      interval: 30s
      timeout: 10s
      retries: 3

  # =============================================================================
  # Custom Application Services
  # =============================================================================
  
  # SOVD Server (Device Endpoint)
  # Simulates vehicle diagnostic system
  # Connects to HONO HTTP adapter as a device
  sovd-server:
    build:
      context: ./python-apps/sovd-server
      dockerfile: Dockerfile
    container_name: sovd-server
    depends_on:
      - hono-adapter-http
    environment:
      # HONO connection configuration
      HONO_HTTP_ADAPTER_HOST: "hono-adapter-http"
      HONO_HTTP_ADAPTER_PORT: "8080"
      
      # Device identification
      HONO_TENANT_ID: "default"
      HONO_DEVICE_ID: "vehicle-001"
      
      # Application configuration
      SOVD_SERVER_PORT: "9090"
      SOVD_LOG_LEVEL: "INFO"
      
      # Diagnostic data simulation
      SOVD_SIMULATION_ENABLED: "true"
      SOVD_SIMULATION_INTERVAL: "30"  # seconds
      
    ports:
      - "9090:9090"  # SOVD server internal port
    networks:
      - hono-network
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:9090/health"]
      interval: 30s
      timeout: 10s
      retries: 3

  # SOVD Client (Business Application)
  # Connects to HONO to communicate with SOVD devices
  # Stores diagnostic data in MongoDB
  sovd-client:
    build:
      context: ./python-apps/sovd-client
      dockerfile: Dockerfile
    container_name: sovd-client
    depends_on:
      - hono-adapter-http
      - mongodb
      - kafka
    environment:
      # HONO connection configuration
      HONO_HTTP_ADAPTER_HOST: "hono-adapter-http"
      HONO_HTTP_ADAPTER_PORT: "8080"
      
      # Target device configuration
      HONO_TENANT_ID: "default"
      HONO_DEVICE_ID: "vehicle-001"
      
      # Kafka configuration for IDS SOC event streaming
      KAFKA_BOOTSTRAP_SERVERS: "kafka:29092"
      KAFKA_TOPIC_TELEMETRY: "hono.telemetry.default"
      KAFKA_TOPIC_EVENTS: "hono.events.default"
      KAFKA_TOPIC_COMMANDS: "hono.commands.default"
      KAFKA_CONSUMER_GROUP: "sovd-client-group"
      
      # MongoDB configuration
      MONGODB_HOST: "mongodb"
      MONGODB_PORT: "27017"
      MONGODB_USERNAME: "admin"
      MONGODB_PASSWORD: "password"
      MONGODB_DATABASE: "sovd_data"
      
      # Application configuration
      SOVD_CLIENT_PORT: "8090"
      SOVD_LOG_LEVEL: "INFO"
      
      # Polling configuration
      SOVD_POLL_INTERVAL: "60"  # seconds
      
    ports:
      - "8090:8090"  # SOVD client web interface
    networks:
      - hono-network
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8090/health"]
      interval: 30s
      timeout: 10s
      retries: 3

  # Device Registration UI
  # Web interface for registering new devices
  # Connects to HONO Device Registry API
  device-registration-ui:
    build:
      context: ./python-apps/registration-ui
      dockerfile: Dockerfile
    container_name: device-registration-ui
    depends_on:
      - hono-device-registry
    environment:
      # HONO Device Registry connection
      HONO_REGISTRY_HOST: "hono-device-registry"
      HONO_REGISTRY_PORT: "8080"
      
      # UI configuration
      UI_PORT: "3000"
      UI_LOG_LEVEL: "INFO"
      
      # Default tenant configuration
      DEFAULT_TENANT_ID: "default"
      
    ports:
      - "3000:3000"  # Registration UI web interface
    networks:
      - hono-network
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:3000/health"]
      interval: 30s
      timeout: 10s
      retries: 3

# =============================================================================
# Network and Volume Configuration
# =============================================================================

networks:
  hono-network:
    driver: bridge
    name: hono-network

volumes:
  # HONO Device Registry persistent storage
  hono-device-registry-data:
    driver: local
    
  # MongoDB persistent storage
  mongodb-data:
    driver: local
    
  # Kafka persistent storage
  kafka-data:
    driver: local
    
  # Zookeeper persistent storage  
  zookeeper-data:
    driver: local
    
  zookeeper-logs:
    driver: local

# =============================================================================
# Configuration Notes
# =============================================================================
#
# 1. All services run in insecure mode (no TLS/authentication)
# 2. Default tenant "default" is used for all devices
# 3. Device Registry API available at http://localhost:28080
# 4. HTTP Adapter available at http://localhost:8080
# 5. Kafka available at localhost:9092
# 6. Registration UI available at http://localhost:3000
# 7. SOVD Client UI available at http://localhost:8090
# 8. MongoDB available at localhost:27017
#
# Kafka Topics (auto-created):
# - hono.telemetry.default (device telemetry data)
# - hono.events.default (device events)
# - hono.commands.default (device commands)
#
# To start the stack:
#   docker-compose up -d
#
# To view logs:
#   docker-compose logs -f [service-name]
#
# To stop the stack:
#   docker-compose down
#
# To rebuild custom applications:
#   docker-compose build [service-name]
#   docker-compose up -d [service-name]
#
# =============================================================================
